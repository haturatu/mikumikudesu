#BDPT用、材質注釈テンプレート
#兼、各パラメータについての説明書

#1.変数の定義
# "型.要素数 : 名前" で材質注釈変数を定義する
# 変数の型はi(32bit整数)またはf(単精度浮動小数) 要素数は1～4

#マテリアルのカテゴリ
i.1 : Category
_E Category : default=0, glass=1, metal=2, sss=3

#_E で列挙子を定義できる。材質注釈内で「Category : default」と書くと「Category : 0」と書いたのと同じ扱いになる
#default 鏡面反射面の下に拡散反射面を持つ。透過を起こさない
#glass 鏡面反射面の下に透過する物質。拡散反射を起こさない。ポリゴンで閉じた空間を作っている必要がある
#metal 鏡面反射面のみを持つ。透過も拡散反射も起こさない
#sss 鏡面反射面の下に表面下散乱を起こす面を持つChristensen-Burleyモデルなのでボリュームは考慮しない)


#光源を作るための情報
f.1 : Light		#Albedoにこの値を乗じた数値の輝度で発光する
f.1 : LightFalloff	#面の法線方向に対して何度の角度まで照射するか360°系で指定。0～90の範囲で指定する
f.1 : LightHotspot	#面の法線方向に対して何度の角度までを減衰無しで照射するか。0～90の範囲で指定する
i.1 : RestrictedLight	#0以外を指定すると光源としてサンプリングされず、周囲を積極的に照明しなくなるが、ポリゴン自体の輝度は上昇する

#BxDFを定義するための情報
f.1 : Roughness	#鏡面反射面の粗さ。xyの2要素。xでtangent方向、yでbinormal方向について指定し、x=yの時、等方性反射になる。0～1の範囲で指定する
f.1 : Anisotropy#粗さの偏り。-1～+1の範囲。0で等方性。+1の時Tangent方向にのみ粗くなり、-1の時Binormal方向にのみ粗くなる。Conty&Kulla[2017]参照
f.2 : IOR	#屈折率。xyの2要素。実際の屈折率は x + y/λ[nm]²で算出され、yは分散を表す。CauchyのABパラメータに相当
f.4 : Cat	#カテゴリ固有パラメータ以下、詳述
#Category == glassの時 xで濃さについての係数。0でAlbedoによらず透明ガラスになる。(1-Albedo)*xに比例した単位長さ当たりの吸収率を持つ
#Category == sssの時 xyzでRGB各チャンネルについての光の平均自由行程、wでスケール。


#テクスチャマッピングのための情報
f.1 : AutoNormal	#モデルに設定されたテクスチャの明度勾配に従う法線マップを自動生成する。値を大きくするほど
f.1 : TextureLoops	#以下の**Loops変数に対する倍率(VolumeLoopsには適用されません)
f.1 : NormalLoops	#法線マップがモデルに設定されたUV1周あたり何周するか
f.1 : NormalScale	#法線マップの影響度の強さ(デコボコ具合) 1より大きくしてもよい
i.1 : NormalUV		#法線のUVをどこから持ってくるか 0:普通のuv座標, 1～4:追加uv座標1～4のxy要素
f.1 : RoughnessLoops	
f.1 : RoughnessScale	#roughnessマップの影響度。0～1で指定する
i.1 : RoughnessUV
f.1 : AlbedoLoops	
f.1 : AlbedoScale	#元の色とのブレンド比。0～1で指定する
i.1 : AlbedoUV
f.1 : AlphaLoops	
f.1 : AlphaScale	#αマップの影響度の強さ。0～1で指定する
i.1 : AlphaUV

#以下、β3から追加
f.1 : Alpha		#不透明度α値への倍率、0を指定するとその材質は表示されない
f.4 : ColorConst	#Albedoを調整するための定数
i.1 : ColorConstOp	#↑をどのように用いるか
_E ColorConstOp : default, add, mul, pow, hsv
# default : 	ColorConst.xyzに設定されたrgb値1色で置き換える
# add : 	ColorConst.xyzに設定されたrgb値と加算する
# mul : 	ColorConst.xyzに設定されたrgb値と乗算する
# pow : 	ColorConst.xyzに設定された係数乗する
# hsv : 	xで色相の変化(加算)、yで彩度の変化(乗算)、zで明度の変化(乗算)
# 共通:		wで影響度(0の時影響無し、1の時以上の操作を行った結果をそのまま採用する)
i.1 : ColorConstUV	#1～4の時、追加UV座標をColorConstにセットする

i.1 : EnableSphereMap	#0でスフィアマップ無効、1で有効
i.1 : AlbedoEstimation	#Albedoの推定法、以下詳述
_E AlbedoEstimation : mmd, diffuse, ambient
#mmd : MMDの標準状態に近い色にする、拡散色*154/255 + 環境色で算出
#diffuse : MMD材質の拡散色をAlbedoとする、
#ambient : MMD材質の環境色をAlbedoとする

i.4 : VolumeDiv		#xyzでUVW方向に何分割されるか、wに総コマ数
f.3 : VolumeLoops	#xyz方向にボリュームテクスチャ何周分を表示するか
i.1 : VolumeWrap	#1:ボリュームテクスチャを敷き詰める。0:テクスチャは1周分だけ使用し範囲外は密度0とみなす



#2.使用可能なテクスチャマップの指定。コメントはテクスチャがどのように使われるか
_T0 : NormalMap		#物体の法線方向を摂動させる
_T1 : RoughnessMap	#Roughness.xyに乗算される
_T2 : AlbedoMap		#Albedo(基本色)を置換する
_T3 : AlphaMap		#不透明度に乗算される



#3.使用可能な3Dテクスチャマップの指定
_V0 : VolumeMap		#Rチャンネルに密度、Gチャンネルに温度を記録



#4.デフォルト値の指定
LightFalloff : 90
LightHotspot : 90
Roughness : 0.5
IOR : 1.5,0
TextureLoops : 1
NormalLoops : 1
NormalScale : 1
RoughnessLoops : 1
RoughnessScale : 1
AlbedoLoops : 1
AlbedoScale : 1
AlphaLoops : 1
AlphaScale : 1
Alpha : 1

AlbedoEstimation : diffuse
EnableSphereMap : 1
VolumeLoops : 1,1,1
VolumeWrap : 1

#余談
#デフォルトではAlbedoEstimationはdiffuse
#mmdを指定すると確かに本家に近い見た目が得られるが白っぽくなりやすい
#また、基本色に関するパラメータが2つある事になるので1からモデルを作る事を考えるとdiffuseの方が直感的であり
#blenderなど最近のDCCツールとの相性も良いと考えられるため
